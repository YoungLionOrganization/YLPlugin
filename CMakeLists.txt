cmake_minimum_required(VERSION 3.26)
project(YLPlugin VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(YLPLUGIN_QT "Build optional Qt RCC adapter" OFF)
option(YLPLUGIN_SANITIZE "Address and undefined behavior sanitizers" OFF)
option(YLPLUGIN_FUZZ "Build Clang libFuzzer reader" OFF)
set(YLPLUGIN_YLSPEC_SOURCE "" CACHE PATH "Existing YLSpec 0.1.2 source (otherwise find installed package)")
if(YLPLUGIN_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(YLPLUGIN_YLSPEC_SOURCE)
  set(YLSPEC_BUILD_CLI OFF CACHE BOOL "" FORCE)
  add_subdirectory("${YLPLUGIN_YLSPEC_SOURCE}" _ylspec)
else()
  find_package(YLSpec 0.1.2 CONFIG REQUIRED)
endif()
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
find_package(nlohmann_json 3.11 CONFIG REQUIRED)
find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(ZSTD_LIBRARY NAMES zstd REQUIRED)
find_package(Threads REQUIRED)
add_library(ylplugin SHARED src/container/container.cpp src/metadata/profile.cpp
  src/crypto/crypto.cpp src/compatibility/preflight.cpp src/runtime/runtime.cpp
  src/container/extract.cpp src/c_api.cpp)
add_library(YLPlugin::ylplugin ALIAS ylplugin)
target_include_directories(ylplugin PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src ${ZSTD_INCLUDE_DIR})
target_link_libraries(ylplugin PRIVATE OpenSSL::Crypto ${ZSTD_LIBRARY} ${CMAKE_DL_LIBS}
  Threads::Threads PUBLIC YLSpec::ylspec nlohmann_json::nlohmann_json)
target_compile_definitions(ylplugin PRIVATE YLPLUGIN_BUILD _FILE_OFFSET_BITS=64)
set_target_properties(ylplugin PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0 WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
  target_compile_options(ylplugin PRIVATE /W4 /permissive-)
else()
  target_compile_options(ylplugin PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_executable(ylplugin_cli tools/cli.cpp)
set_target_properties(ylplugin_cli PROPERTIES OUTPUT_NAME ylplugin)
target_link_libraries(ylplugin_cli PRIVATE YLPlugin::ylplugin OpenSSL::Crypto)
if(YLPLUGIN_QT)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Core)
  add_library(ylplugin_qt adapters/qt/rcc.cpp)
  add_library(YLPlugin::qt ALIAS ylplugin_qt)
  target_include_directories(ylplugin_qt PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_link_libraries(ylplugin_qt PUBLIC YLPlugin::ylplugin Qt6::Core)
  install(TARGETS ylplugin_qt EXPORT YLPluginTargets)
endif()
if(BUILD_TESTING)
  add_executable(ylplugin_tests tests/core.cpp)
  target_link_libraries(ylplugin_tests PRIVATE ylplugin OpenSSL::Crypto Threads::Threads)
  target_compile_definitions(ylplugin_tests PRIVATE YLP_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME plugin_core COMMAND ylplugin_tests)
  add_executable(ylplugin_c_example examples/read.c)
  target_link_libraries(ylplugin_c_example PRIVATE ylplugin)
  add_library(ylplugin_native_fixture MODULE tests/native_fixture.c)
  target_include_directories(ylplugin_native_fixture PRIVATE include)
  add_test(NAME native_lifecycle COMMAND ylplugin_tests --native $<TARGET_FILE:ylplugin_native_fixture>)
  set_tests_properties(native_lifecycle PROPERTIES SKIP_RETURN_CODE 77)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME cli_and_python COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:ylplugin_cli> $<TARGET_FILE:ylplugin>)
  add_test(NAME conformance_vectors COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/conformance.py $<TARGET_FILE:ylplugin>)
  if(UNIX)
    add_test(NAME cli_signal_cancel COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_cancel.py $<TARGET_FILE:ylplugin_cli>)
  endif()
  if(YLPLUGIN_SANITIZE AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    execute_process(COMMAND ${CMAKE_CXX_COMPILER} -print-file-name=libasan.so OUTPUT_VARIABLE ylp_asan OUTPUT_STRIP_TRAILING_WHITESPACE)
    execute_process(COMMAND ${CMAKE_CXX_COMPILER} -print-file-name=libstdc++.so OUTPUT_VARIABLE ylp_stdcpp OUTPUT_STRIP_TRAILING_WHITESPACE)
    set_tests_properties(cli_and_python conformance_vectors PROPERTIES ENVIRONMENT
      "LD_PRELOAD=${ylp_asan}:${ylp_stdcpp};ASAN_OPTIONS=detect_leaks=0:halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1")
  endif()
  if(YLPLUGIN_QT)
    add_executable(ylplugin_qt_tests tests/qt.cpp)
    target_link_libraries(ylplugin_qt_tests PRIVATE ylplugin_qt)
    target_compile_definitions(ylplugin_qt_tests PRIVATE YLP_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME qt_rcc_lifecycle COMMAND ylplugin_qt_tests)
    find_package(Qt6 6.4 QUIET COMPONENTS Qml Quick)
    if(TARGET Qt6::Qml AND TARGET Qt6::Quick)
      add_executable(ylplugin_qml_tests tests/qml.cpp)
      target_link_libraries(ylplugin_qml_tests PRIVATE ylplugin_qt Qt6::Qml Qt6::Quick)
      target_compile_definitions(ylplugin_qml_tests PRIVATE YLP_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
      add_test(NAME qml_engine_lifecycle COMMAND ylplugin_qml_tests)
      set_tests_properties(qml_engine_lifecycle PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    endif()
  endif()
endif()
if(YLPLUGIN_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "YLPLUGIN_FUZZ requires Clang")
  endif()
  target_compile_options(ylplugin PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  target_link_options(ylplugin PRIVATE -fsanitize=address,undefined)
  add_executable(ylplugin_fuzz tests/fuzz.cpp)
  target_link_libraries(ylplugin_fuzz PRIVATE ylplugin)
  target_compile_options(ylplugin_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(ylplugin_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
install(TARGETS ylplugin ylplugin_cli EXPORT YLPluginTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT YLPluginTargets NAMESPACE YLPlugin:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/YLPlugin)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/YLPluginConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/YLPluginConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/YLPlugin)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/YLPluginConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/YLPluginConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/YLPluginConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/YLPlugin)
install(DIRECTORY docs/ schemas/ DESTINATION ${CMAKE_INSTALL_DATADIR}/ylplugin)
