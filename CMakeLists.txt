# Bindings keep embedded crypto/core symbols private, avoiding collisions with
# a host (such as Node) that already has another OpenSSL build loaded.
foreach(ylp_binding_target ylplugin_node ylplugin_jni ylplugin_runtime)
  if(TARGET ${ylp_binding_target})
    set_target_properties(${ylp_binding_target} PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
    if(UNIX AND NOT APPLE)
      target_link_options(${ylp_binding_target} PRIVATE "LINKER:--exclude-libs,ALL")
    elseif(APPLE)
      set(ylp_exports "${CMAKE_CURRENT_BINARY_DIR}/${ylp_binding_target}.exports")
      if(ylp_binding_target STREQUAL "ylplugin_node")
        file(WRITE "${ylp_exports}" "_napi_register_module_v1\n_node_api_module_get_api_version_v1\n")
      elseif(ylp_binding_target STREQUAL "ylplugin_jni")
        file(WRITE "${ylp_exports}" "_Java_xyz_younglion_ylplugin_Native_open\n_Java_xyz_younglion_ylplugin_Native_close\n_Java_xyz_younglion_ylplugin_Native_invoke\n")
      else()
        file(WRITE "${ylp_exports}" "")
        file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/include/ylplugin/ylplugin.h" ylp_declarations REGEX "^YLP_API ")
        foreach(ylp_declaration IN LISTS ylp_declarations)
          string(REGEX MATCH "(ylp_[a-z_]+)\\(" ylp_match "${ylp_declaration}")
          if(ylp_match)
            file(APPEND "${ylp_exports}" "_${CMAKE_MATCH_1}\n")
          endif()
        endforeach()
      endif()
      target_link_options(${ylp_binding_target} PRIVATE "LINKER:-exported_symbols_list,${ylp_exports}")
    endif()
  endif()
endforeach()
